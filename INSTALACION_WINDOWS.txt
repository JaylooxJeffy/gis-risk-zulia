## PASO 6: FRONTEND - Modificar auth.js
# Abre: frontend/auth.js
# Busca la función redirectByRole (línea ~366)
# Reemplaza toda la función con esto:

function redirectByRole(rol) {
    sessionStorage.setItem('userRole', rol);
    
    // Verificar si la contraseña es temporal
    const user = ApiClient.getUser();
    if (user && user.password_temporal === true) {
        window.location.href = 'change-password.html';
        return;
    }
    
    window.location.href = 'sig-zulia-pro.html';
}
